import os
import asyncio
import random
from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup
from telegram.ext import Application, CommandHandler, CallbackQueryHandler, ContextTypes

# ==================== ESTADO DO JOGO ====================
games = {}

class RachaGame:
    def __init__(self, chat_id):
        self.chat_id = chat_id
        self.players = {}  # user_id: {dados do jogador}
        self.state = "LOBBY"
        self.turn = 1
        self.max_turns = 5

# ==================== OBSTÁCULOS E CENÁRIOS ALEATÓRIOS (RP) ====================

OBSTACULOS_PISTA = [
    "🚨 Um carro da Polícia surge do nada com a sirene ligada cruzando o canteiro central!",
    "🐕 Um cachorro vira-lata caramelo atravessa a pista correndo atrás de um gato!",
    "🛢️ Um caminhão de lixo deixou cair uma poça de óleo derramado no meio da curva!",
    "🚦 O sinal do cruzamento fecha e um ônibus de viagem cruza a pista em alta velocidade!",
    "🪨 Houve um deslizamento de pedras e entulho no acostamento!",
    "👵 Uma velhinha atravessa a faixa de pedestres bem no trecho de aceleração!",
    "🌧️ Começa uma chuva torrencial repentina, deixando o asfalto igual a uma pista de gelo!",
    "📦 Um engradado de cerveja cai da carroceria de uma picape bem na sua frente!"
]

SUCESSOS_MANOBRA = [
    "rasgou o asfalto, trocou de marcha com maestria e passou voando!",
    "deu um cavalo de pau perfeito, desviou do perigo e disparou na frente!",
    "segurou o volante com fúria, fez a curva cantando pneu e ganhou vantagem!",
    "jogou o carro de lado na tangência e costurou o trânsito com pura habilidade!",
    "pisou no fundo sem dó e fez os adversários comerem poeira!"
]

ACIDENTES_BATER = [
    "tentou dar uma de velozes e furiosos, mas perdeu a traseira e enrolou o carro num poste!",
    "não viu o obstáculo, cravou os freios tarde demais e rodou bonito na pista!",
    "forçou demais a troca de marcha e o bloco do motor simplesmente EXPLODIU em chamas!",
    "calculou mal a curva, bateu de quina na mureta de proteção e destruiu a suspensão!",
    "foi tentar desviar de última hora, subiu no meio-fio e capotou três vezes!"
]

# ==================== COMANDOS DO BOT ====================

async def start_game(update: Update, context: ContextTypes.DEFAULT_TYPE):
    chat_id = update.effective_chat.id
    
    if chat_id in games and games[chat_id].state == "RUNNING":
        await update.message.reply_text("🎀 *Marie:* 'Calma aí, piloto! Já tem um Racha da Marie pegando fogo aqui perto!'", parse_mode="Markdown")
        return

    games[chat_id] = RachaGame(chat_id)
    
    keyboard = [[InlineKeyboardButton("🏎️ Colar no Grid de Largada", callback_data="join_game")]]
    reply_markup = InlineKeyboardMarkup(keyboard)

    await update.message.reply_text(
        "🎀 **A MARIE SUBIU NO CAPÔ E ANUNCIOU:**\n\n"
        "_'Atenção, pilotos! O **Racha da Marie** vai começar agora!_\n"
        "_'Vocês têm **30 segundos** para alinhar os carros no grid!'_\n\n"
        "📌 Clique no botão abaixo para colocar seu nome na lista da corrida.",
        reply_markup=reply_markup,
        parse_mode="Markdown"
    )

    await asyncio.sleep(30)
    await run_game(chat_id, context)


async def handle_buttons(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    chat_id = query.message.chat_id
    user = query.from_user
    user_id = user.id
    data = query.data

    await query.answer()

    if chat_id not in games:
        return

    game = games[chat_id]

    # Cria a menção direta com o username ou nome do jogador
    mencao_user = user.mention_markdown_v2()

    # --- ENTRAR NO LOBBY ---
    if data == "join_game":
        if game.state != "LOBBY":
            await query.answer("Os motores já aceleraram! Você perdeu a largada da Marie.", show_alert=True)
            return

        if user_id in game.players:
            await query.answer("Seu carro já está alinhado na pista!", show_alert=True)
            return

        game.players[user_id] = {
            "name": user.first_name,
            "mention": mencao_user,
            "distancia": 0,
            "quebrado": False,
            "escolheu": False,
            "ultima_acao": None
        }
        await query.message.reply_text(
            f"🚘 O piloto {mencao_user} alinhou a sua máquina na pista do **Racha da Marie** 🎀 e deu duas cutucadas no acelerador!",
            parse_mode="MarkdownV2"
        )

    # --- ESCOLHA DE AÇÕES ---
    elif data.startswith("action_"):
        if game.state != "RUNNING":
            return

        action = data.split("_")[1]
        
        if user_id not in game.players:
            await query.answer("Você está assistindo do acostamento, não está correndo!", show_alert=True)
            return

        player = game.players[user_id]

        if player["quebrado"]:
            await query.answer("Seu carro virou sucata nesta corrida! Você só assiste.", show_alert=True)
            return

        if player["escolheu"]:
            await query.answer("Você já decidiu sua manobra nesta rodada!", show_alert=True)
            return

        player["escolheu"] = True
        player["ultima_acao"] = action
        await query.answer("Manobra engatada! Aguarde a narração da Marie...")


# ==================== LÓGICA E NARRAÇÃO DO JOGO ====================

async def run_game(chat_id: int, context: ContextTypes.DEFAULT_TYPE):
    game = games.get(chat_id)
    if not game:
        return

    if len(game.players) < 2:
        await context.bot.send_message(
            chat_id=chat_id,
            text="🎀 *Marie:* 'Sem plateia e com menos de 2 pilotos? O Racha da Marie foi cancelado!'\n\n*(Mínimo de 2 jogadores)*",
            parse_mode="Markdown"
        )
        del games[chat_id]
        return

    game.state = "RUNNING"
    await context.bot.send_message(
        chat_id=chat_id,
        text="🎀 **A MARIE DROPA A BANDEIRA DE LARGADA!** 🚨\n\n"
             "O Racha da Marie começou! O ronco dos motores ecoa na noite! **É TUDO OU NADA!**",
        parse_mode="Markdown"
    )

    # Loop de 5 Rodadas
    for turn in range(1, game.max_turns + 1):
        game.turn = turn
        
        for p in game.players.values():
            p["escolheu"] = False
            p["ultima_acao"] = None

        keyboard = [
            [
                InlineKeyboardButton("🔥 Morder a Traseira (Acelerar)", callback_data="action_acelerar"),
                InlineKeyboardButton("🧠 Trajetória Perfeita (Cautela)", callback_data="action_seguro")
            ],
            [
                InlineKeyboardButton("⚡ Injetar Nitro (Tudo ou Nada)", callback_data="action_nitro")
            ]
        ]
        reply_markup = InlineKeyboardMarkup(keyboard)

        # Sorteia um obstáculo aleatório para a rodada inteira
        obstaculo_da_rodada = random.choice(OBSTACULOS_PISTA)

        await context.bot.send_message(
            chat_id=chat_id,
            text=f"📜 **[ RACHA DA MARIE 🎀 — TRECHO {turn}/{game.max_turns} ]**\n\n"
                 f"⚠️ **IMPREVISTO NA PISTA:** {obstaculo_da_rodada}\n\n"
                 f"⚙️ *Pilotos, vocês têm 15 segundos para decidir a manobra no painel!*",
            reply_markup=reply_markup,
            parse_mode="Markdown"
        )

        await asyncio.sleep(15)

        # Relatório Narrado com marcação dos players
        relatorio = f"🎀 **NARRAÇÃO DA MARIE - TRECHO {turn}:**\n"
        relatorio += "───────────────────────────\n\n"

        for p_id, p in game.players.items():
            if p["quebrado"]:
                continue

            acao = p["ultima_acao"]
            mencao = p["mention"]  # Marcação do usuário

            if acao is None:
                p["distancia"] += 10
                relatorio += f"🥱 O piloto {mencao} moscou na largada, hesitou na manobra e avançou só **10 metros**.\n\n"

            elif acao == "seguro":
                p["distancia"] += 30
                relatorio += f"🎯 O piloto {mencao} jogou com a cabeça, manteve a cautela e ganhou **30 metros** em segurança.\n\n"

            elif acao == "acelerar":
                # 25% de chance aleatória de bater
                if random.random() < 0.25:
                    p["quebrado"] = True
                    acidente = random.choice(ACIDENTES_BATER)
                    relatorio += f"💥 O piloto {mencao} {acidente} **(FORA DO RACHA DA MARIE!)**\n\n"
                else:
                    p["distancia"] += 50
                    sucesso = random.choice(SUCESSOS_MANOBRA)
                    relatorio += f"⚙️ O piloto {mencao} {sucesso} **(+50 metros)**\n\n"

            elif acao == "nitro":
                # 50% de chance aleatória de dar ruim
                if random.random() < 0.50:
                    p["quebrado"] = True
                    acidente = random.choice(ACIDENTES_BATER)
                    relatorio += f"🔥 O piloto {mencao} acionou o nitro, mas deu ruim: {acidente} **(MOTOR EXPLODIU!)**\n\n"
                else:
                    p["distancia"] += 90
                    relatorio += f"⚡ O piloto {mencao} ATIVOU O NITRO E VOOU NA PISTA! **(+90 metros de vantagem!)**\n\n"

        await context.bot.send_message(chat_id=chat_id, text=relatorio, parse_mode="MarkdownV2")
        await asyncio.sleep(3)

    # ==================== CLASSIFICAÇÃO FINAL ====================
    ranking = sorted(game.players.values(), key=lambda x: x["distancia"], reverse=True)

    texto_final = "🏁 **A LINHA DE CHEGADA DO RACHA DA MARIE 🎀 FOI CRUZADA!** 🏁\n\n"
    
    posicao = 1
    for p in ranking:
        mencao = p["mention"]
        if p["quebrado"]:
            status_desc = "💥 *Perda Total (Guincho)*"
        else:
            status_desc = f"🏎️ *{p['distancia']} metros percorridos*"

        if posicao == 1 and not p["quebrado"]:
            texto_final += f"👑 **1º LUGAR:** {mencao}\n┗ status: {status_desc} — *Campeão do Racha da Marie 🎀 e Dono do Asfalto!*\n\n"
        else:
            texto_final += f"🏎️ **{posicao}º LUGAR:** {mencao}\n┗ status: {status_desc}\n\n"
            
        posicao += 1

    await context.bot.send_message(chat_id=chat_id, text=texto_final, parse_mode="MarkdownV2")
    del games[chat_id]


# ==================== INICIALIZAÇÃO ====================

def main():
    TOKEN = os.getenv("TELEGRAM_TOKEN", "COLE_SEU_TOKEN_AQUI_SE_FOR_RODAR_LOCAL")

    app = Application.builder().token(TOKEN).build()

    app.add_handler(CommandHandler("criarracha", start_game))
    app.add_handler(CallbackQueryHandler(handle_buttons))

    print("🎀 Racha da Marie...")
    app.run_polling()

if __name__ == "__main__":
    main()
